<script>


let tiempo = 600;
let activo = true;


function actualizar(){

let minutos=Math.floor(tiempo/60);
let segundos=tiempo%60;


let m=String(minutos).padStart(2,'0');
let s=String(segundos).padStart(2,'0');


let pantalla=document.getElementById("timer");


pantalla.innerHTML=
m+'<span class="dot">:</span>'+s;



if(tiempo<=60){

pantalla.classList.add("warning");

}



if(tiempo<=0){

pantalla.innerHTML="00:00";

pantalla.classList.add("finish");

activo=false;

return;

}



tiempo--;

setTimeout(actualizar,1000);

}



function reiniciar(){

tiempo=600;

activo=true;


let pantalla=document.getElementById("timer");

pantalla.classList.remove("warning");
pantalla.classList.remove("finish");


actualizar();

}



// inicia automáticamente

actualizar();



</script>
